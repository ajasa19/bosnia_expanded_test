# Albania receives the invitation (ROOT = Albania, FROM = Bosnia)
country_event = {
    id = bosnia.200
    title = bosnia.200.t
    desc = bosnia.200.d
    picture = GFX_report_event_generic_handshake

    is_triggered_only = yes
    fire_only_once = yes

    option = {
        name = bosnia.200.a
        ai_chance = { base = 50 }
        FROM = {
            add_to_faction = ROOT
            give_guarantee = ROOT
        }
        every_other_country = {
            limit = {
                is_in_faction_with = FROM
                NOT = { tag = FROM }
            }
            give_guarantee = ROOT
        }

        if = {
            limit = { country_exists = ITA }
            ITA = {
                add_ai_strategy = {
                    type = conquer
                    id = ALB
                    value = -200
                }
                if = {
                    limit = { has_wargoal_against = ALB }
                    remove_wargoal = {
                        type = all
                        target = ALB
                    }
                }
            }
        }
    }

    option = {
        name = bosnia.200.b
        ai_chance = { base = 50 }
        FROM = {
            country_event = { id = bosnia.201 hours = 1 }
        }
    }
}
